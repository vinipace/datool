import * as React from "react";
import { Card, CardHeader, CardTitle, CardDescription, CardContent, Button, Input, Select, Notice, Tabs, TabsList, TabsTrigger, TabsContent, DataTable } from "@datool/ui";
import { ChartContainer, ChartTooltip, ChartTooltipContent, BarChart, Bar, XAxis, YAxis, CartesianGrid } from "@datool/charts";
const categories: Record<string, string> = { content: "Conteúdo", technical: "Técnico", external_authority: "Autoridade externa" };
const confidence: Record<string, string> = { high: "Alta", medium: "Média", low: "Baixa" };
const list = (value: unknown): any[] => Array.isArray(value) ? value : [];
const decode = (value: unknown): any => { try { return typeof value === "string" ? JSON.parse(value) : value; } catch { return null; } };
function Evidence({ fact }: { fact: any }) {
  return <article className="space-y-3 border-b border-border py-4"><div className="flex flex-wrap gap-2 text-xs text-foreground-muted"><span>{fact.factType}</span><span>Confiança {confidence[fact.confidence]}</span></div><p className="text-sm">{fact.statement}</p>{list(fact.evidenceQuotes).map((quote, i) => <blockquote key={i} className="border-l-2 border-border pl-3 text-sm text-foreground-muted">{quote.excerpt}<code className="mt-1 block break-all text-xs">{quote.evidenceRef}</code></blockquote>)}{list(fact.limitations).map((text, i) => <p key={i} className="text-xs text-foreground-muted">{text}</p>)}</article>;
}
export default function Diagnosis({ trace }: ViewProps) {
  const [query, setQuery] = React.useState("");
  const [category, setCategory] = React.useState("all");
  const [level, setLevel] = React.useState("all");
  const [layout, setLayout] = React.useState("cards");
  const output = decode(trace.output), input = decode(trace.input);
  const actions = React.useMemo(() => list(output?.finalActions), [output]);
  const facts = list(output?.investigation?.facts);
  const opportunities = list(output?.investigation?.opportunities);
  const order = list(output?.actionPlan?.orderedActionIds);
  const rank = (action: any) => { const i = order.indexOf(action.actionId); return i < 0 ? action.priority : i + 1; };
  const rows = actions.filter(a => (category === "all" || a.category === category) && (level === "all" || a.confidence === level) && [a.title, a.deliverable, a.problem].join(" ").toLowerCase().includes(query.toLowerCase())).map(a => ({ ...a, rank: rank(a) })).sort((a,b) => a.rank - b.rank);
  const columns = [
    { accessorKey: "rank", header: "Ordem" },
    { accessorKey: "title", header: "Ação" },
    { accessorKey: "category", header: "Categoria", cell: ({getValue}: {getValue: () => string}) => categories[getValue()] },
    { accessorKey: "confidence", header: "Confiança", cell: ({getValue}: {getValue: () => string}) => confidence[getValue()] },
  ];
  if (output?.kind !== "review_dossier") return <main className="p-4"><Notice title={trace.status === "running" ? "Diagnóstico em andamento" : "Dossier indisponível"}>{output?.truncated ? "Este trace antigo contém um resultado truncado. Abra uma execução com o resultado completo." : "Esta view apresenta o output review_dossier do workflow Evidence to action."}</Notice></main>;
  const chartData = Object.entries(categories).map(([id, name]) => ({ name, count: actions.filter(a => a.category === id).length }));
  const withheld = list(output.actionPlan?.actions).filter(a => !actions.some(f => f.actionId === a.actionId));
  return <main className="mx-auto max-w-6xl space-y-5 p-4 text-foreground sm:p-6">
    <header className="space-y-2"><p className="text-xs font-medium uppercase tracking-widest text-foreground-muted">Evidência → Diagnóstico</p><h1 className="text-2xl font-semibold">{input?.briefing?.brand?.canonicalName || "Visibilidade em IA"}</h1><p className="text-sm text-foreground-muted">{input?.promptSegments?.[0]?.name?.split(" · ")[0]} · {list(input?.snapshot?.runs).length} respostas</p></header>
    <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">{[[actions.length,"Ações finais"],[withheld.length,"Rascunhos retidos"],[opportunities.filter(o => o.status === "supported").length,"Oportunidades sustentadas"],[facts.length,"Fatos documentados"]].map(([value,label]) => <Card key={label}><CardHeader><CardDescription>{label}</CardDescription><CardTitle className="text-3xl tabular-nums">{value}</CardTitle></CardHeader><CardContent /></Card>)}</div>
    <Notice variant={output.audit?.planVerdict === "approved" ? "success" : "warning"} title={output.audit?.planVerdict === "approved" ? "Plano aprovado pela auditoria do modelo" : "Plano requer revisão"}>Confiança representa suporte da evidência, não probabilidade de sucesso. Revisão humana ainda é necessária.</Notice>
    <Card><CardHeader><CardTitle>Ações por categoria</CardTitle><CardDescription>Contagem de ações finais; categorias sem evidência suficiente podem ter zero.</CardDescription></CardHeader><CardContent><ChartContainer config={{count:{label:"Ações", color:"var(--data-series-1)"}}} className="h-40 w-full"><BarChart data={chartData} accessibilityLayer><CartesianGrid vertical={false}/><XAxis dataKey="name" tickLine={false} axisLine={false} interval={0} tick={{fontSize:11}}/><YAxis allowDecimals={false} width={24} tickLine={false} axisLine={false}/><ChartTooltip content={<ChartTooltipContent/>}/><Bar dataKey="count" fill="var(--color-count)" radius={6} isAnimationActive={false}/></BarChart></ChartContainer></CardContent></Card>
    <Tabs defaultValue="actions" className="space-y-4"><TabsList className="flex-wrap"><TabsTrigger value="actions">Ações</TabsTrigger><TabsTrigger value="opportunities">Oportunidades</TabsTrigger><TabsTrigger value="evidence">Evidências</TabsTrigger><TabsTrigger value="audit">Auditoria</TabsTrigger></TabsList>
      <TabsContent value="actions" className="space-y-4"><div className="grid gap-2 sm:grid-cols-3"><Input aria-label="Buscar ações" placeholder="Buscar ações…" value={query} onChange={e=>setQuery(e.target.value)}/><Select aria-label="Categoria" value={category} onChange={e=>setCategory(e.target.value)}><option value="all">Todas as categorias</option>{Object.entries(categories).map(([id,label])=><option key={id} value={id}>{label}</option>)}</Select><Select aria-label="Confiança" value={level} onChange={e=>setLevel(e.target.value)}><option value="all">Todas as confianças</option>{Object.entries(confidence).map(([id,label])=><option key={id} value={id}>{label}</option>)}</Select></div><div className="flex gap-2"><Button size="sm" variant={layout === "cards" ? "secondary" : "ghost"} aria-pressed={layout === "cards"} onClick={()=>setLayout("cards")}>Cards</Button><Button size="sm" variant={layout === "table" ? "secondary" : "ghost"} aria-pressed={layout === "table"} onClick={()=>setLayout("table")}>Tabela</Button></div><p className="text-sm text-foreground-muted">{output.actionPlan?.rankingRationale}</p>
      {layout === "table" ? <DataTable data={rows} columns={columns} caption="Ações priorizadas" emptyMessage="Nenhuma ação corresponde aos filtros."/> : rows.map(action=><Card key={action.actionId}><CardHeader><p className="text-xs text-foreground-muted">#{action.rank} · {categories[action.category]} · Confiança {confidence[action.confidence]}</p><CardTitle>{action.title}</CardTitle><CardDescription>{action.objective}</CardDescription></CardHeader><CardContent className="mt-4 space-y-4"><p className="text-sm">{action.deliverable}</p><p className="text-xs text-foreground-muted">Responsável: {action.ownerRole} · Esforço: {action.effort}</p><details className="border-t border-border pt-3"><summary className="cursor-pointer text-sm font-medium">O que fazer</summary><ul className="ml-4 list-disc space-y-2 py-3 text-sm">{list(action.changeDetails).map((text,i)=><li key={i}>{text}</li>)}</ul><p className="text-sm text-foreground-muted">{action.mechanism}</p></details><details className="border-t border-border pt-3"><summary className="cursor-pointer text-sm font-medium">Evidências vinculadas</summary>{facts.filter(f=>list(action.factIds).includes(f.factId)).map(f=><Evidence key={f.factId} fact={f}/>)}</details><details className="border-t border-border pt-3"><summary className="cursor-pointer text-sm font-medium">Pré-requisitos e validação</summary><ul className="ml-4 list-disc space-y-2 py-3 text-sm">{list(action.prerequisites).map((text,i)=><li key={i}>{text}</li>)}</ul>{list(Object.values(action.validationPlan||{})).map((text,i)=><p key={i} className="my-2 text-sm text-foreground-muted">{text}</p>)}</details></CardContent></Card>)}{layout === "cards" && !rows.length && <Notice>Nenhuma ação corresponde aos filtros. Veja a auditoria para decisões e rascunhos retidos.</Notice>}</TabsContent>
      <TabsContent value="opportunities" className="space-y-3">{opportunities.map(o=><Card key={o.opportunityId}><CardHeader><CardDescription>{categories[o.category]} · {o.status}</CardDescription><CardTitle>{o.problem}</CardTitle></CardHeader><CardContent className="mt-3 space-y-3 text-sm"><p>{o.controllableLever}</p><p className="text-foreground-muted">{o.whyMaterial}</p>{list(o.missingEvidence).map((text,i)=><Notice key={i} variant="warning">{text}</Notice>)}</CardContent></Card>)}</TabsContent>
      <TabsContent value="evidence">{facts.map(f=><Evidence key={f.factId} fact={f}/>)}</TabsContent>
      <TabsContent value="audit" className="space-y-4">{list(output.categoryDecisions).map(c=><Card key={c.category}><CardHeader><CardTitle>{categories[c.category]}</CardTitle><CardDescription>{c.status}</CardDescription></CardHeader><CardContent className="mt-3 text-sm">{c.reason}</CardContent></Card>)}{withheld.map(a=><Notice key={a.actionId} variant="warning" title={a.title}>Rascunho retido — não é uma ação final.</Notice>)}{list(output.audit?.actionReviews).map(r=><Card key={r.actionId}><CardHeader><CardTitle>{r.actionId} · {r.verdict}</CardTitle></CardHeader><CardContent className="mt-3 text-sm"><p>{r.reasoning}</p>{list(r.requiredChanges).map((text,i)=><p key={i} className="my-2">{text}</p>)}</CardContent></Card>)}<h2 className="font-semibold">Limitações</h2><ul className="ml-4 list-disc space-y-2 text-sm text-foreground-muted">{list(output.reviewSummary?.limitations).map((text,i)=><li key={i}>{text}</li>)}</ul></TabsContent>
    </Tabs>
  </main>;
}
